    {
      "@type": "Question",
      "name": "How do ArmorNGlory EVA foam clogs fit?",
      "acceptedAnswer": {
        "@type": "Answer",
        "text": "ArmorNGlory EVA foam clogs have a roomy comfort fit with a pivoting heel strap. If between sizes or preferring a closer fit, size down."
      }
    }
  ]

</script>
